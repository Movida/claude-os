# L'archive de démarrage : ce que le noyau déballe dans son système de
# fichiers (voir archive.sh pour le format, celui de gen_init_cpio). Ce
# qu'il lance ensuite, et dans quel ordre, c'est /etc/demarrage qui le
# dit. Les chemins sources sont relatifs au dossier de l'étape (make les
# construit d'abord).

# Les programmes, à root, exécutables par tous.
dir   /bin 755 0 0
file  /bin/bonjour.wasm    bonjour/target/wasm32-unknown-unknown/release/bonjour.wasm 755 0 0
file  /bin/lecteur.wasm    lecteur/target/wasm32-unknown-unknown/release/lecteur.wasm 755 0 0
file  /bin/bonjour-asm     linux/construits/bonjour-asm 755 0 0
file  /bin/pile            linux/construits/pile 755 0 0
file  /bin/sse             linux/construits/sse 755 0 0
file  /bin/bonjour-c       linux/construits/bonjour-c 755 0 0
file  /bin/bonjour-c-pie   linux/construits/bonjour-c-pie 755 0 0
file  /bin/bonjour-c-dynamique linux/construits/bonjour-c-dynamique 755 0 0
file  /bin/dynamique       linux/construits/dynamique 755 0 0
file  /bin/memoire         linux/construits/memoire 755 0 0
file  /bin/claude-version  linux/construits/claude-version 755 0 0
file  /bin/temps           linux/construits/temps 755 0 0
# Celui-ci est à l'utilisateur 1000 : il vérifie qu'il ne peut pas écrire
# son propre binaire pendant qu'il tourne (ETXTBSY), ce que les droits
# d'un fichier de root refuseraient avant (EACCES).
file  /bin/fichiers        linux/construits/fichiers 755 1000 1000
file  /bin/arbre           linux/construits/arbre 755 0 0
file  /bin/segments        linux/construits/segments 755 0 0
file  /bin/signaux         linux/construits/signaux 755 0 0
file  /bin/systeme         linux/construits/systeme 755 0 0
file  /bin/evenements      linux/construits/evenements 755 0 0
file  /bin/proc            linux/construits/proc 755 0 0

# Les commandes que lance le noyau (formats/src/commande.rs). Écrit avant
# les autres fichiers de /etc : un dossier se lit du plus récent au plus
# ancien (6b), lecteur.wasm le verra donc en dernier, après ceux qu'il
# s'attend à trouver.
dir   /etc 755 0 0
file  /etc/demarrage       racine/etc/demarrage 644 0 0
# Où ld-linux cherche les bibliothèques (ld.so.conf), et ce qu'il y a
# trouvé, fabriqué par ldconfig (le cache : voir le Makefile).
file  /etc/ld.so.conf      racine/etc/ld.so.conf 644 0 0
file  /etc/ld.so.cache     glibc/racine/etc/ld.so.cache 644 0 0

# Quelques fichiers à lire : un texte, un second nom (un lien physique),
# et des liens symboliques.
file  /etc/bonjour.txt     racine/etc/bonjour.txt 644 0 0 /etc/salut.txt
slink /etc/lien-bonjour    bonjour.txt 777 0 0
# Un lien absolu : un programme WebAssembly ne doit pas pouvoir le suivre
# (il sortirait de son dossier pré-ouvert, voir noyau/src/wasi.rs).
slink /etc/lien-absolu     /etc/bonjour.txt 777 0 0

# /tmp, ouvert à tous mais « sticky » : chacun n'y efface que ses fichiers.
# Un fichier de root s'y trouve, que l'utilisateur 1000 ne doit pas pouvoir
# effacer, bien qu'il puisse l'écrire (le test de linux/fichiers).
dir   /tmp 1777 0 0
file  /tmp/a-root          racine/etc/bonjour.txt 666 0 0
# Le dossier personnel de l'utilisateur 1000, celui des programmes (HOME).
dir   /home 755 0 0
dir   /home/utilisateur 755 1000 1000

# La glibc partagée de la machine (6c), copiée par le Makefile dans
# glibc/ (jamais commitée), aux chemins d'Ubuntu : les fichiers dans
# /usr/lib/x86_64-linux-gnu, et /lib, /lib64 des liens vers /usr (le
# « /usr fusionné »). Le PT_INTERP des programmes dynamiques,
# /lib64/ld-linux-x86-64.so.2, passe ainsi par deux liens symboliques.
dir   /usr 755 0 0
dir   /usr/lib 755 0 0
dir   /usr/lib/x86_64-linux-gnu 755 0 0
file  /usr/lib/x86_64-linux-gnu/ld-linux-x86-64.so.2 glibc/racine/usr/lib/x86_64-linux-gnu/ld-linux-x86-64.so.2 755 0 0
file  /usr/lib/x86_64-linux-gnu/libc.so.6 glibc/racine/usr/lib/x86_64-linux-gnu/libc.so.6 755 0 0
# Les quatre autres bibliothèques de claude, avec les droits d'Ubuntu.
file  /usr/lib/x86_64-linux-gnu/libm.so.6 glibc/racine/usr/lib/x86_64-linux-gnu/libm.so.6 644 0 0
file  /usr/lib/x86_64-linux-gnu/libpthread.so.0 glibc/racine/usr/lib/x86_64-linux-gnu/libpthread.so.0 644 0 0
file  /usr/lib/x86_64-linux-gnu/libdl.so.2 glibc/racine/usr/lib/x86_64-linux-gnu/libdl.so.2 644 0 0
file  /usr/lib/x86_64-linux-gnu/librt.so.1 glibc/racine/usr/lib/x86_64-linux-gnu/librt.so.1 644 0 0
dir   /usr/lib64 755 0 0
slink /usr/lib64/ld-linux-x86-64.so.2 ../lib/x86_64-linux-gnu/ld-linux-x86-64.so.2 777 0 0
slink /lib                 usr/lib 777 0 0
slink /lib64               usr/lib64 777 0 0
# Une bibliothèque à nous, là où s'installe ce qui ne vient pas de la
# distribution : linux/dynamique la charge par dlopen, ld-linux la trouve
# par son cache (/usr/local/lib est dans ld.so.conf).
dir   /usr/local 755 0 0
dir   /usr/local/lib 755 0 0
file  /usr/local/lib/libessai.so linux/construits/libessai.so 644 0 0
